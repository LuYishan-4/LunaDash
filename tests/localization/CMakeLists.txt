cmake_minimum_required(VERSION 3.21)
project(LunaDashLocalizationTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core)
get_filename_component(REPOSITORY_ROOT ${CMAKE_CURRENT_LIST_DIR}/../.. ABSOLUTE)
add_library(ludash-localization ${REPOSITORY_ROOT}/src/config/localization/Localization.cpp
    ${REPOSITORY_ROOT}/src/config/localization/JsonTranslator.cpp)
target_include_directories(ludash-localization PUBLIC ${REPOSITORY_ROOT}/src)
target_link_libraries(ludash-localization PUBLIC Qt6::Core)
qt_add_resources(ludash-localization translations PREFIX /LuDash BASE ${REPOSITORY_ROOT}
    FILES ${REPOSITORY_ROOT}/data/translations/en_US.json ${REPOSITORY_ROOT}/data/translations/zh_TW.json)
include(${REPOSITORY_ROOT}/cmake/LunaDashTranslations.cmake)
add_executable(localization-test main.cpp)
target_link_libraries(localization-test PRIVATE ludash-localization)
enable_testing()
add_test(NAME localization-resources COMMAND localization-test ${REPOSITORY_ROOT}/data/translations)
